Esta guía te muestra cómo crear un flujo de trabajo de pedido completo, desde la creación del cliente hasta la entrega del pedido.

<div id="creating-and-processing-an-order">
  ## Creación y processing de un pedido
</div>

<Steps>
  <Step title="Obtén tu clave de la API">
    [Crea una clave de API en tu panel](https://plantstore.com/settings/api-keys), que usarás para autenticar tus solicitudes.

    Guárdala de forma segura y pásala al SDK como variable de entorno mediante un archivo `.env`, o directamente en la configuración de tu aplicación.

    ```bash title=".env"
    PLANT_STORE_API_KEY=your_api_key_here
    ```
  </Step>

  <Step title="Instalar el SDK (Kit de desarrollo de software)">
    También usaremos la biblioteca `dotenv` para cargar nuestra clave de API desde una variable de entorno.

    <CodeBlocks>
      ```bash Python
      pip install plantstore-sdk
      pip install python-dotenv
      ```

      ```bash TypeScript
      npm install @plantstore/sdk
      npm install dotenv
      ```

      ```bash Go
      go get github.com/plantstore/plantstore-go
      ```
    </CodeBlocks>
  </Step>

  <Step title="Crear un flujo de pedido completo">
    Crea un nuevo archivo llamado `create_order.py`, `create_order.ts` o `main.go` y agrega el siguiente código:

    <CodeBlocks>
      ```python Python
      # create_order.py
      import os
      from dotenv import load_dotenv
      from plantstore import PlantStoreClient

      load_dotenv()

      client = PlantStoreClient(
          api_key=os.getenv("PLANT_STORE_API_KEY")
      )

      # Crear un cliente
      customer = client.customers.create(
          email="gardener@example.com",
          name="Jane Smith",
          phone="+1-555-0123"
      )

      print(f"Cliente creado: {customer.id}")

      # Buscar plantas disponibles
      plants = client.plants.search(status="available")
      selected_plant = plants[0]

      print(f"Planta seleccionada: {selected_plant.name}")

      # Crear el pedido
      order = client.orders.create(
          customer_id=customer.id,
          items=[
              {
                  "plant_id": selected_plant.id,
                  "quantity": 2,
                  "price": selected_plant.price
              }
          ],
          shipping_address={
              "street": "123 Garden Ave",
              "city": "Portland",
              "state": "OR",
              "zip": "97201"
          }
      )

      print(f"Pedido creado: {order.id}")

      # Procesar pago
      payment = client.orders.process_payment(
          order_id=order.id,
          payment_method="card",
          amount=order.total
      )

      print(f"Pago procesado: {payment.status}")

      # Actualizar estado del pedido a processing
      updated_order = client.orders.update(
          order_id=order.id,
          status="processing"
      )

      print(f"Estado del pedido: {updated_order.status}")
      print(f"¡Pedido completado! ID: {order.id}")
      ```

      ```typescript TypeScript
      // create_order.ts
      import { PlantStoreClient } from '@plantstore/sdk';
      import 'dotenv/config';

      const client = new PlantStoreClient({
        apiKey: process.env.PLANT_STORE_API_KEY
      });

      async function createOrder() {
        // Crear un cliente
        const customer = await client.customers.create({
          email: 'gardener@example.com',
          name: 'Jane Smith',
          phone: '+1-555-0123'
        });

        console.log(`Cliente creado: ${customer.id}`);

        // Explorar plantas disponibles
        const plants = await client.plants.search({ status: 'available' });
        const selectedPlant = plants[0];

        console.log(`Planta seleccionada: ${selectedPlant.name}`);

        // Crear el pedido
        const order = await client.orders.create({
          customerId: customer.id,
          items: [
            {
              plantId: selectedPlant.id,
              quantity: 2,
              price: selectedPlant.price
            }
          ],
          shippingAddress: {
            street: '123 Garden Ave',
            city: 'Portland',
            state: 'OR',
            zip: '97201'
          }
        });

        console.log(`Pedido creado: ${order.id}`);

        // Procesar el pago
        const payment = await client.orders.processPayment({
          orderId: order.id,
          paymentMethod: 'card',
          amount: order.total
        });

        console.log(`Pago procesado: ${payment.status}`);

        // Actualizar el estado del pedido a "processing"
        const updatedOrder = await client.orders.update({
          orderId: order.id,
          status: 'processing'
        });

        console.log(`Estado del pedido: ${updatedOrder.status}`);
        console.log(`¡Pedido completado! ID: ${order.id}`);
      }

      createOrder();
      ```

      ```go Go
      // main.go
      package main

      import (
          "fmt"
          "log"
          "os"

          "github.com/joho/godotenv"
          "github.com/plantstore/plantstore-go"
      )

      func main() {
          godotenv.Load()

          client := plantstore.NewClient(
              plantstore.WithAPIKey(os.Getenv("PLANT_STORE_API_KEY")),
          )

          // Crear un cliente
          customer, err := client.Customers.Create(&plantstore.CustomerCreateRequest{
              Email: "gardener@example.com",
              Name:  "Jane Smith",
              Phone: "+1-555-0123",
          })
          if err != nil {
              log.Fatal(err)
          }

          fmt.Printf("Cliente creado: %s\n", customer.ID)

          // Buscar plantas disponibles
          plants, err := client.Plants.Search(&plantstore.PlantSearchRequest{
              Status: "available",
          })
          if err != nil {
              log.Fatal(err)
          }
          selectedPlant := plants[0]

          fmt.Printf("Planta seleccionada: %s\n", selectedPlant.Name)

          // Crear el pedido
          order, err := client.Orders.Create(&plantstore.OrderCreateRequest{
              CustomerID: customer.ID,
              Items: []plantstore.OrderItem{
                  {
                      PlantID:  selectedPlant.ID,
                      Quantity: 2,
                      Price:    selectedPlant.Price,
                  },
              },
              ShippingAddress: plantstore.Address{
                  Street: "123 Garden Ave",
                  City:   "Portland",
                  State:  "OR",
                  Zip:    "97201",
              },
          })
          if err != nil {
              log.Fatal(err)
          }

          fmt.Printf("Pedido creado: %s\n", order.ID)

          // Procesar el pago
          payment, err := client.Orders.ProcessPayment(&plantstore.PaymentRequest{
              OrderID:       order.ID,
              PaymentMethod: "card",
              Amount:        order.Total,
          })
          if err != nil {
              log.Fatal(err)
          }

          fmt.Printf("Pago procesado: %s\n", payment.Status)

          // Actualizar el estado del pedido
          updatedOrder, err := client.Orders.Update(&plantstore.OrderUpdateRequest{
              OrderID: order.ID,
              Status:  "processing",
          })
          if err != nil {
              log.Fatal(err)
          }

          fmt.Printf("Estado del pedido: %s\n", updatedOrder.Status)
          fmt.Printf("¡Pedido completado! ID: %s\n", order.ID)
      }
      ```
